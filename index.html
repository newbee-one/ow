/* 拒绝采样，避免取模偏差；无 Web Crypto 时退回普通随机数。 */

function randomInteger(max){

  if(!Number.isInteger(max)||max<1) throw new Error("Invalid random range");

  if(!globalThis.crypto?.getRandomValues) return Math.floor(Math.random()*max);

  const limit = Math.floor(4294967296/max)*max;

  const buffer = new Uint32Array(1);

  do{crypto.getRandomValues(buffer)}while(buffer[0]>=limit);

  return buffer[0]%max;

}


/* 每次避开上一款，其余酒款均可被抽取。 */

function boundedText(value,maxLength){

  return typeof value==="string" &&

    value.trim().length>0 &&

    Array.from(value.trim()).length<=maxLength;

}


async function requestAI(entry){

  if(!AI_CONFIG.enabled) return {entry,usedFallback:false};


  const controller = new AbortController();

  const timer = setTimeout(()=>controller.abort(),AI_CONFIG.timeoutMs);


  try{

    if(!AI_CONFIG.endpoint) throw new Error("AI endpoint is not configured");


    const endpoint = new URL(AI_CONFIG.endpoint,window.location.href);

    if(!["https:","http:"].includes(endpoint.protocol)){

      throw new Error("Unsupported endpoint");

    }


    const response = await fetch(endpoint.href,{

      method:"POST",

      headers:{"Content-Type":"application/json"},

      signal:controller.signal,

      credentials:"same-origin",

      body:JSON.stringify({

        date:DATE.full,

        selectedCocktail:{

          id:entry.id,

          name:entry.name,

          englishName:entry.englishName

        },

        fallback:{advice:entry.advice,quote:entry.quote}

      })

    });


    if(!response.ok) throw new Error("AI request failed");

    const data = await response.json();


    if(!boundedText(data.advice,9)||!/^[宜忌]\s/.test(data.advice)){

      throw new Error("Invalid advice");

    }

    if(!boundedText(data.quote,32)) throw new Error("Invalid quote");


    const lines = data.quote.trim().split(/\r?\n/);

    if(lines.length>2||lines.some(line=>Array.from(line).length>16)){

      throw new Error("Quote exceeds display size");

    }

    if(!boundedText(data.note,50)||!boundedText(data.mood,18)){

      throw new Error("Invalid pairing");

    }


    return {

      entry:{

        ...entry,

        advice:data.advice.trim(),

        quote:data.quote.trim(),

        note:data.note.trim(),

        mood:data.mood.trim(),

        credit:"今日短句 · AI生成"

      },

      usedFallback:false

    };

  }catch(error){

    /* 网络失败时仍能正常玩，使用本地匹配内容。 */

    return {entry,usedFallback:true};

  }finally{

    clearTimeout(timer);

  }

}


async function spin(){

  if(busy) return;

  setBusy(true);

  $("lever").classList.add("pulled");

  $("machine").classList.add("rolling");

  $("status").textContent = "宜忌转动，一句话正在赶来……";


  const chosen = nextIndex(currentIndex);

  let tick = 0;

  let interval = null;


  if(!reduceMotion()){

    interval = setInterval(()=>{

      renderMachine(DATABASE[(chosen+(tick++))%DATABASE.length]);

    },95);

  }


  try{

    const [result] = await Promise.all([

      requestAI({...DATABASE[chosen]}),

      sleep(reduceMotion()?100:1200)

    ]);

    currentIndex = chosen;

    current = result.entry;

    $("status").textContent = result.usedFallback

      ? "AI 暂不可用，已抽取本地内容。按中间按钮留下这一签。"

      : `今日「${current.advice}」 · 按中间按钮，把这一签留下。`;

  }catch(error){

    $("status").textContent = "这次没转起来，请再试一次。";

  }finally{

    if(interval!==null) clearInterval(interval);

    renderMachine(current);

    $("machine").classList.remove("rolling");

    $("lever").classList.remove("pulled");

    setBusy(false);

  }

}

  if(busy||event.button!==0) return;

  pointerStart = event.clientY;

  dragged = false;

  $("lever").setPointerCapture(event.pointerId);

});


$("lever").addEventListener("pointermove",event=>{

  if(pointerStart===null||busy) return;

  const distance = Math.max(0,event.clientY-pointerStart);

  $("lever").style.transform = `rotate(${Math.min(35,distance*.5)}deg)`;

  if(distance>25) dragged = true;

});


$("lever").addEventListener("pointerup",()=>{

  if(pointerStart===null) return;

  pointerStart = null;

  $("lever").style.transform = "";

  if(dragged){

    ignoreClickUntil = Date.now()+500;

    spin();

  }

});


$("lever").addEventListener("pointercancel",()=>{

  pointerStart = null;

  dragged = false;

  $("lever").style.transform = "";

});


$("lever").addEventListener("lostpointercapture",()=>{

  pointerStart = null;

  $("lever").style.transform = "";

});


$("lever").addEventListener("click",()=>{

  if(Date.now()<ignoreClickUntil) return;

  spin();

});

/* 根据真实文字宽度换行，保存长酒名和中文时不裁切。 */

function wrapText(ctx,text,maxWidth){

  const result = [];

  for(const paragraph of text.split("\n")){

    let line = "";

    for(const character of Array.from(paragraph)){

      const candidate = line+character;

      if(line&&ctx.measureText(candidate).width>maxWidth){

        result.push(line);

        line = character;

      }else{

        line = candidate;

      }

    }

    result.push(line);

  }

  return result;

}


function drawFitted(ctx,text,y,startSize,maxWidth,family){

  let size = startSize;

  ctx.font = `${size}px ${family}`;

  while(size>18&&ctx.measureText(text).width>maxWidth){

    size--;

    ctx.font = `${size}px ${family}`;

  }

  ctx.fillText(text,500,y);

}
